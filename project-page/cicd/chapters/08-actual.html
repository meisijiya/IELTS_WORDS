deploy.yml

文件：.github/workflows/deploy.yml。 关键差异：默认 permissions: contents: read；build job 在 push 才跑；deploy job 显式做 DB 保护（不允许 down -v 或 --force-recreate postgres）。

build-and-push job

仅 push 触发，runner = ubuntu-latest。

jobs:
  build-and-push:
    runs-on: ubuntu-latest
    if: github.event_name == 'push'

    steps:
      - uses: actions/checkout@v4
      - uses: docker/setup-buildx-action@v3
      - uses: docker/login-action@v3
        with:
          registry: ${{ secrets.ALIYUN_REGISTRY }}
          username: ${{ secrets.ALIYUN_REGISTRY_USERNAME }}
          password: ${{ secrets.ALIYUN_REGISTRY_PASSWORD }}

      - name: Extract version
        id: meta
        run: |
          echo "short_sha=$(echo ${{ github.sha }} | cut -c1-7)" >> $GITHUB_OUTPUT
          echo "timestamp=$(date +%Y%m%d-%H%M%S)" >> $GITHUB_OUTPUT

      - uses: docker/build-push-action@v6
        with:
          context: .
          push: true
          build-args: |
            AUDIO_BUNDLE_URL=https://github.com/meisijiya/IELTS_WORDS/releases/download/audio-full-2026-07-23/audio_full.tgz
          tags: |
            ${{ secrets.ALIYUN_REGISTRY }}/${{ secrets.ALIYUN_REGISTRY_NAMESPACE }}/yasi-words:latest
            ${{ secrets.ALIYUN_REGISTRY }}/${{ secrets.ALIYUN_REGISTRY_NAMESPACE }}/yasi-words:${{ steps.meta.outputs.short_sha }}
            ${{ secrets.ALIYUN_REGISTRY }}/${{ secrets.ALIYUN_REGISTRY_NAMESPACE }}/yasi-words:${{ steps.meta.outputs.timestamp }}
          # cache 故意关闭: 避免 stale layer 掩盖 AUDIO_BUNDLE_URL 改动
          cache-from: ""
          cache-to: ""

deploy job（核心是 DB 保护 + APP_IMAGE 显式 export）

  deploy:
    runs-on: ubuntu-latest
    needs: build-and-push
    if: github.event_name == 'push'

    steps:
      - uses: appleboy/ssh-action@v1
        with:
          host: ${{ secrets.DEPLOY_HOST }}
          username: ${{ secrets.DEPLOY_SSH_USER || 'root' }}
          key: ${{ secrets.DEPLOY_SSH_KEY }}
          script: |
            set -euo pipefail
            cd /opt/yasi-words

            # DB 保护: postgres 必须先在跑；否则 abort
            docker ps --filter "name=yasi-postgres" --filter "status=running" \
              --format "{{.Names}}" | grep -q '^yasi-postgres$' || {
              echo "[deploy] FATAL: yasi-postgres not running. ABORT."
              exit 1
            }

            export ALIYUN_REGISTRY='${{ secrets.ALIYUN_REGISTRY }}'
            export ALIYUN_REGISTRY_NAMESPACE='${{ secrets.ALIYUN_REGISTRY_NAMESPACE }}'

            echo "${{ secrets.ALIYUN_REGISTRY_PASSWORD }}" | docker login \
              -u '${{ secrets.ALIYUN_REGISTRY_USERNAME }}' \
              --password-stdin ${{ secrets.ALIYUN_REGISTRY }}

            export APP_IMAGE='${{ secrets.ALIYUN_REGISTRY }}/${{ secrets.ALIYUN_REGISTRY_NAMESPACE }}/yasi-words:latest'
            docker pull "$APP_IMAGE"
            docker compose pull app
            # 只 recreate app，不动 postgres 与其 volume
            docker compose up -d --force-recreate app

            # 保留当前 + 1 个最近非当前 image，清理其余
            CURRENT_ID=$(docker ps --filter "name=yasi-app" --format "{{.Image}}" | head -1)
            docker images --format "{{.ID}}|{{.CreatedAt}}|{{.Repository}}:{{.Tag}}" \
              | grep "yasi-words" | sort -t'|' -k2 -r \
              | awk -F'|' -v keep_id="$CURRENT_ID" \
                  '($1 == keep_id) {next} (++seen > 1) {print $1}' \
              | xargs -r docker rmi -f 2>&1 | tail -5 || true
            docker image prune -f --filter "until=24h" || true
            docker builder prune -f --filter "until=24h" || true

            # DB 保护: post-check
            if ! docker exec yasi-postgres pg_isready \
                 -U "$(grep '^POSTGRES_USER=' /opt/yasi-words/.env | cut -d= -f2-)" \
                 -d "$(grep '^POSTGRES_DB=' /opt/yasi-words/.env | cut -d= -f2-)" \
                 > /dev/null 2>&1; then
              echo "[deploy] WARN: pg_isready failed"
            fi
            df -h / | tail -1

Health check

docker-compose.yml

关键点：app service 同时有 image: 和 build:，靠 ${APP_IMAGE:-} 切换 deploy 模式 vs 本地 dev。 audio_data named volume 持久化 entrypoint 拉下来的 audio。 

services:
  postgres:
    image: postgres:16-alpine
    container_name: yasi-postgres
    restart: unless-stopped
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-yasi}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-yasi_dev_pwd_change_me}
      POSTGRES_DB: ${POSTGRES_DB:-yasi_db}
      TZ: ${TZ:-Asia/Shanghai}
    volumes:
      - pgdata:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-yasi} -d ${POSTGRES_DB:-yasi_db}"]
      interval: 5s
      timeout: 5s
      retries: 10
    networks: [yasi-net]

  app:
    image: ${APP_IMAGE:-}     # deploy 时 = ACR tag；本地 dev 留空走 build:
    build:
      context: .
      dockerfile: Dockerfile
      args:
        AUDIO_BUNDLE_URL: ${AUDIO_BUNDLE_URL:-}
    container_name: yasi-app
    restart: unless-stopped
    depends_on:
      postgres:
        condition: service_healthy
    environment:
      DATABASE_URL: postgresql://${POSTGRES_USER:-yasi}:${POSTGRES_PASSWORD:-yasi_dev_pwd_change_me}@postgres:5432/${POSTGRES_DB:-yasi_db}
      ADMIN_PASSWORD: ${ADMIN_PASSWORD:-<admin-password>}
      SESSION_SECRET: ${SESSION_SECRET:-change-me-in-production-min-32-chars-long}
      NODE_ENV: production
      TZ: ${TZ:-Asia/Shanghai}
    ports:
      - "${APP_PORT:-3000}:3000"
    volumes:
      - app_logs:/app/logs
      - audio_data:/app/public/audio

Dockerfile

两阶段 build。builder 阶段加 mkdir -p /app/public 解决 坑 5；runner 阶段 COPY --from=builder /app/public/ ./public/ 必须路径存在。 

FROM node:22-alpine AS builder
RUN sed -i 's|dl-cdn.alpinelinux.org|mirrors.aliyun.com|g' /etc/apk/repositories
ENV npm_config_registry=https://registry.npmmirror.com
WORKDIR /app
RUN apk add --no-cache python3 py3-pip curl
COPY package.json package-lock.json ./
RUN npm ci
COPY . .
RUN mkdir -p /app/public        # 解决坑 5
ARG AUDIO_BUNDLE_URL=""
RUN if [ -n "$AUDIO_BUNDLE_URL" ]; then \
      mkdir -p public/audio; \
      curl -fsSL --max-time 600 "$AUDIO_BUNDLE_URL" \
        | tar xz --strip-components=1 -C public/audio/ \
        || echo "WARN: audio fetch failed"; \
    fi
RUN npx prisma generate
RUN npm run build

FROM node:22-alpine AS runner
RUN sed -i 's|dl-cdn.alpinelinux.org|mirrors.aliyun.com|g' /etc/apk/repositories
ENV npm_config_registry=https://registry.npmmirror.com
WORKDIR /app
ENV NODE_ENV=production
RUN apk add --no-cache python3 py3-pip curl
COPY package.json package-lock.json ./
COPY --from=builder /app/.next ./.next
COPY --from=builder /app/node_modules ./node_modules
COPY --from=builder /app/prisma ./prisma
COPY --from=builder /app/seed ./seed
COPY --from=builder /app/src ./src
COPY --from=builder /app/tools ./tools
COPY --from=builder /app/public/ ./public/
COPY docker/entrypoint.sh /usr/local/bin/entrypoint.sh
RUN chmod +x /usr/local/bin/entrypoint.sh
EXPOSE 3000
HEALTHCHECK --interval=30s --timeout=10s --start-period=20s --retries=3 \
  CMD curl -fsS http://localhost:3000/login >/dev/null || exit 1
ENTRYPOINT ["/usr/local/bin/entrypoint.sh"]
CMD ["node", "node_modules/next/dist/bin/next", "start", "-H", "0.0.", "-p", "3000"]

entrypoint.sh

三件事：等 DB → push schema → seed；audio 缺失时 runtime fetch（兜底）。 

#!/bin/sh
set -eu
echo "[entrypoint] starting Yasi Words container..."

# 1. 按 DATABASE_URL 切换 Prisma provider
if echo "$DATABASE_URL" | grep -qE '^postgres'; then
  if grep -q 'provider = "sqlite"' prisma/schema.prisma; then
    cp prisma/schema.prisma prisma/schema.sqlite.prisma.bak
    sed -i 's/provider = "sqlite"/provider = "postgresql"/' prisma/schema.prisma
    npx prisma generate
    NEED_RESTORE=1
  fi
fi

# 2. 等 Postgres 30s
i=0
while [ "$i" -lt 30 ]; do
  if node -e "const {PrismaClient}=require('@prisma/client');const p=new PrismaClient();p.\$queryRaw\`SELECT 1\`.then(()=>{p.\$disconnect();process.exit(0);}).catch(()=>process.exit(1));" >/dev/null 2>&1; then
    break
  fi
  i=$((i+1)); sleep 1
done
[ "$i" -ge 30 ] && { echo "[entrypoint] DB not ready"; exit 1; }

# 3. push schema（loud-exit，无 || true）
npx prisma db push --skip-generate --accept-data-loss 2>&1 | tail -5 || { echo "[entrypoint] db push failed"; exit 1; }

# 4. seed
if [ -d "seed" ]; then
  npx tsx prisma/seed.ts 2>&1 | tail -30 || { echo "[entrypoint] seed failed"; exit 1; }
fi

# 5. audio 兜底: baked 缺失时 runtime 拉
AUDIO_COUNT=$(find public/audio -type f -name "*.mp3" 2>/dev/null | wc -l)
if [ "$AUDIO_COUNT" -eq 0 ] && [ -n "$AUDIO_BUNDLE_URL" ]; then
  mkdir -p public/audio
  curl -fsSL --retry 3 --retry-delay 5 --max-time 600 "$AUDIO_BUNDLE_URL" \
    | tar xz --strip-components=1 -C public/audio/ \
    && AUDIO_COUNT=$(find public/audio -type f -name "*.mp3" | wc -l)
fi

# 6. 复原 schema
[ "${NEED_RESTORE:-0}" = "1" ] && mv prisma/schema.sqlite.prisma.bak prisma/schema.prisma

exec "$@"

耦合关系

改一处必看另外三处：.github/workflows/deploy.yml ↔ docker-compose.yml ↔ docker/entrypoint.sh ↔ Dockerfile。
  • 改 AUDIO_BUNDLE_URL：deploy.yml build-args + Dockerfile ARG + entrypoint runtime fetch 三处都改。
  • 改 Postgres 主版本：Dockerfile 不变；docker-compose.yml image: + entrypoint 都不引用具体版本（仅 Prisma provider 改动）。
  • 改 container name：deploy.yml DB 保护 grep 字符串 + docker-compose.yml container_name: 同步。
  • 改 server 路径：deploy.yml cd /opt/yasi-words + 所有 ops workflow cd 同步（详见 坑 11）。

后续改动请保持这 4 个文件的耦合关系一起改。这是本项目跑通的核心。
